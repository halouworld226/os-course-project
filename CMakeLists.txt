cmake_minimum_required(VERSION 3.16)
project(os_starter LANGUAGES CXX)

find_package(Threads REQUIRED)
add_executable(os_server src/server.cpp src/net.cpp src/image.cpp src/worker.cpp src/scheduler.cpp src/job_scheduler.cpp src/storage.cpp src/inode_fs.cpp)
target_link_libraries(os_server PRIVATE Threads::Threads)
target_compile_features(os_server PRIVATE cxx_std_17)
set_target_properties(os_server PROPERTIES CXX_EXTENSIONS OFF)
if(WIN32)
    target_link_libraries(os_server PRIVATE ws2_32)
endif()
if(MSVC)
    target_compile_options(os_server PRIVATE /W4)
else()
    target_compile_options(os_server PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME upload_process_download
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" "$<TARGET_FILE:os_server>" --tcp)
    add_test(NAME process_pipeline
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" "$<TARGET_FILE:os_server>" --stdio)
    set_tests_properties(upload_process_download process_pipeline PROPERTIES TIMEOUT 45)
    add_test(NAME protocol_boundaries COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_check.py" "$<TARGET_FILE:os_server>")
    add_test(NAME web_routes COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/web_smoke.py" "$<TARGET_FILE:os_server>")
    add_test(NAME session_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_check.py" "$<TARGET_FILE:os_server>")
    add_test(NAME web_persistent COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/web_smoke.py" "$<TARGET_FILE:os_server>" --session)
    add_executable(scheduler_check tests/scheduler_check.cpp src/scheduler.cpp src/worker.cpp src/image.cpp)
    add_executable(shared_scheduler_check tests/shared_scheduler_check.cpp src/job_scheduler.cpp src/worker.cpp src/image.cpp)
    add_executable(storage_check tests/storage_check.cpp src/storage.cpp src/inode_fs.cpp src/image.cpp)
    add_executable(worker_handle_check tests/worker_handle_check.cpp src/worker.cpp src/image.cpp)
    foreach(check scheduler_check shared_scheduler_check storage_check worker_handle_check)
        target_compile_features(${check} PRIVATE cxx_std_17)
        target_include_directories(${check} PRIVATE src)
        target_link_libraries(${check} PRIVATE Threads::Threads)
    endforeach()
    add_test(NAME scheduler_recovery COMMAND scheduler_check)
    add_test(NAME worker_handle_isolation COMMAND worker_handle_check)
    add_test(NAME priority_shared_pool COMMAND shared_scheduler_check)
    add_test(NAME inode_storage COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/storage_runner.py" "$<TARGET_FILE:storage_check>")
endif()
